Path::extension()

The part of the filename after the last dot, or None if there is no dot other than a leading dot.

Example
Path{ p: "README" }.extension() //-> None
Path{ p: "archive.tar.gz" }.extension() //-> Some("gz")
Path{ p: ".gitignore" }.extension() //-> None
Path{ p: ".config.json" }.extension() //-> Some("json")
Path{ p: "file." }.extension() //-> Some("")
Source Code
public method extension(this: Path): Option<String> {
  let name = match this.file_name() {
    Some(name) => name
    None => return None
  }

  if name == ".." {
    return None
  }

  let parts = name.strip_prefix(".").split(".")
  if parts.len() < 2 {
    return None
  }

  parts.last()
}
